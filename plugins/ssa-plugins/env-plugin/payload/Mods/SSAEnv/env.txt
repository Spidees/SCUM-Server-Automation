# SSAEnv - environment tuning. Re-read every few seconds, no restart needed.
#
# Every value is a MULTIPLIER of the game's own default:  1.0 = unchanged, 2.0 = double,
# 0.5 = half. The originals are recorded automatically the first time the mod runs, into
# ..\SSAEnv.defaults.txt, and every multiplier is applied to those - so editing this file
# never compounds and you never have to write a default down yourself.
#
# Multipliers above 50 are capped: past that you are not tuning the weather simulation,
# you are breaking it.

enabled = 1                      # 0 = put the game's own values back and stop changing anything

drying_multiplier = 3.0          # >1 = players and clothes dry faster
wetting_multiplier = 0.5         # <1 = get wet slower (water immersion + wet surfaces)
rain_multiplier = 0.5            # <1 = get wet slower from RAIN
dirtiness_multiplier = 0.3       # <1 = get dirty slower; 0 = never get dirty from surfaces
surface_wetness_multiplier = 1.0 # <1 = wet ground adds less wetness

debug = 0                        # 1 = write diagnostics to the UE4SS log; 0 = silent

# ── Removing the mod ─────────────────────────────────────────────────────────
# You usually do NOT need to do anything: part of what this mod changes lives in a shared
# game asset that is reloaded from scratch on every server start, so a restart already
# puts the game back to normal.
#
# Only if you want it back to normal WITHOUT restarting: set enabled = 0 above, wait ~10
# seconds, and then disable or remove the mod. (restore = 1 does the same thing and is
# kept for older setups.)
restore = 0
